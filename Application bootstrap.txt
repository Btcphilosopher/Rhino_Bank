import sys

from PySide6.QtWidgets import QApplication

from app.application import RhinoApplication


def main() -> int:

    qt = QApplication(sys.argv)

    qt.setApplicationName("RhinoBank")
    qt.setApplicationDisplayName("RhinoBank Institutional Terminal")
    qt.setOrganizationName("RhinoBank")

    application = RhinoApplication(qt)

    application.start()

    return qt.exec()


if __name__ == "__main__":
    raise SystemExit(main())
2. Application controller
# app/application.py

from PySide6.QtCore import QObject

from app.window import RhinoMainWindow
from core.account import AccountService
from core.trading import TradingService
from core.risk import RiskService
from core.escrow import EscrowService
from core.settlement import SettlementService
from analytics.r_engine import REngine
from infrastructure.health import HealthMonitor


class RhinoApplication(QObject):

    def __init__(self, qt_application):

        super().__init__()

        self.qt = qt_application

        self.account = AccountService()
        self.trading = TradingService()
        self.risk = RiskService()
        self.escrow = EscrowService()
        self.settlement = SettlementService()

        self.analytics = REngine()

        self.health = HealthMonitor()

        self.window = RhinoMainWindow(
            services=self
        )


    def start(self):

        self.health.start()

        self.window.show()

        self.window.status_message(
            "RHINOBANK // TERMINAL INITIALIZED"
        )


    def shutdown(self):

        self.health.stop()

        self.analytics.shutdown()

        self.trading.shutdown()

        self.settlement.shutdown()

6. R integration

The wrapper can launch R as a separate analytics process rather than embedding R directly into the Qt event loop.

# analytics/r_engine.py

import subprocess
import json
from pathlib import Path


class REngine:

    def __init__(self):

        self.rscript = "Rscript"

        self.script_directory = Path(
            "analytics/r"
        )


    def execute(
        self,
        script,
        payload
    ):

        process = subprocess.run(

            [
                self.rscript,

                str(
                    self.script_directory /
                    script
                ),

                json.dumps(
                    payload
                )

            ],

            capture_output=True,

            text=True,

            check=False

        )

        if process.returncode != 0:

            raise RuntimeError(
                process.stderr
            )

        return process.stdout


    def shutdown(self):

        pass

This is preferable to making the Qt application depend directly on an R runtime for every UI operation.

The key architectural idea

The Python wrapper becomes the orchestrator:

                         RHINOBANK.EXE
                              │
                     PYTHON APPLICATION
                              │
          ┌───────────────────┼───────────────────┐
          │                   │                   │
          ▼                   ▼                   ▼
     Qt/PySide6          Python Core          R Analytics
     GUI SHELL           Trading             Statistics
          │                   │                   │
          │                   ├── Orders         ├── Sharpe
          │                   ├── Positions      ├── VaR
          │                   ├── Escrow         ├── Drawdown
          │                   ├── Settlement     ├── Volatility
          │                   └── Risk           └── Performance
          │
          ▼
     CHART / TERMINAL
          │
          ▼
       USER
